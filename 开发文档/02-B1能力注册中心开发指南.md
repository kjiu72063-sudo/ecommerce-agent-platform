## 一、B1 目标

实现能力注册中心（Registry），管理所有定义对象的元数据、版本、状态和关联关系。

### 1.1 核心职责

- Tool/Skill/Prompt/Policy 的 CRUD 操作
- 版本管理与依赖解析
- 状态机迁移引擎
- 内容摘要计算与校验
- 查询与检索

### 1.2 不包含

- 运行时状态管理（B2）
- Context 组装（B3）
- Agent 执行（B4）

### 任务 1：数据库设计（第 1-2 天）

#### 2.1.1 表结构设计

```sql
-- 定义对象主表
CREATE TABLE definition_objects (
    id VARCHAR(64) PRIMARY KEY,          -- 如 agt_xxx
    kind VARCHAR(32) NOT NULL,           -- AgentSpec/SkillManifest/...
    key VARCHAR(128) NOT NULL,           -- 人类可读标识
    namespace VARCHAR(128) NOT NULL,     -- 命名空间
    version VARCHAR(32) NOT NULL,        -- 语义版本 1.0.0
    revision INTEGER NOT NULL DEFAULT 1, -- 乐观并发修订号
    scope_type VARCHAR(32) NOT NULL,     -- system/tenant/user/project
    tenant_id VARCHAR(64),
    user_id VARCHAR(64),
    project_id VARCHAR(64),
    content_digest VARCHAR(70) NOT NULL, -- sha256:xxx
    spec JSONB NOT NULL,                 -- 完整 spec 内容
    status JSONB NOT NULL,               -- 状态信息
    labels JSONB DEFAULT '{}',
    annotations JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL,
    created_by JSONB NOT NULL,           -- ActorRef
    UNIQUE(namespace, key, version)
);

-- 版本历史表
CREATE TABLE definition_versions (
    id SERIAL PRIMARY KEY,
    object_id VARCHAR(64) NOT NULL REFERENCES definition_objects(id),
    version VARCHAR(32) NOT NULL,
    revision INTEGER NOT NULL,
    content_digest VARCHAR(70) NOT NULL,
    spec_snapshot JSONB NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    created_by JSONB NOT NULL
);

-- 依赖关系表
CREATE TABLE definition_dependencies (
    id SERIAL PRIMARY KEY,
    source_id VARCHAR(64) NOT NULL REFERENCES definition_objects(id),
    source_version VARCHAR(32) NOT NULL,
    target_id VARCHAR(64) NOT NULL REFERENCES definition_objects(id),
    target_version VARCHAR(32) NOT NULL,
    dependency_type VARCHAR(32) NOT NULL, -- prompt/context/model/loop/permission/skill/tool
    UNIQUE(source_id, source_version, target_id, dependency_type)
);
```

#### 2.1.2 索引设计

```sql
-- 常用查询索引
CREATE INDEX idx_definition_kind ON definition_objects(kind);
```

**学习要点**：

### 任务 2：Repository 层（第 3-4 天）

#### 2.2.1 定义 Repository 接口

```python
# B1-能力注册中心/src/registry/repository.py

from abc import ABC, abstractmethod
from typing import Optional, List
from dataclasses import dataclass

@dataclass
class DefinitionFilter:
    """定义对象查询过滤器"""
    kind: Optional[str] = None
    namespace: Optional[str] = None
    key: Optional[str] = None
    phase: Optional[str] = None
    tenant_id: Optional[str] = None
    
class DefinitionRepository(ABC):
    """定义对象仓储接口"""
    
    @abstractmethod
    async def create(self, obj: dict) -> str:
        """创建定义对象，返回 ID"""
        pass
    
    @abstractmethod
    async def get_by_id(self, id: str) -> Optional[dict]:
        """按 ID 获取"""
        pass
    
    @abstractmethod
    async def get_by_nkv(self, namespace: str, key: str, version: str) -> Optional[dict]:
        """按 namespace + key + version 获取"""
        pass
    
    @abstractmethod
    async def update(self, id: str, expected_revision: int, updates: dict) -> bool:
        """乐观并发更新"""
        pass
    
    @abstractmethod
    async def list_by_filter(self, filter: DefinitionFilter, limit: int = 100) -> List[dict]:
        """按过滤器查询"""
        pass
    
    @abstractmethod
    async def get_dependencies(self, id: str, version: str) -> List[dict]:
        """获取依赖关系"""
        pass
```

**学习要点**：
- Repository 模式分离数据访问逻辑
- 乐观并发通过 revision 字段实现
- 接口设计支持后续数据库切换

### 任务 3：服务层（第 5-7 天）

#### 2.3.1 Registry 服务

```python
# B1-能力注册中心/src/registry/service.py

from agent_platform_contracts.models import RESOURCE_MODELS
from agent_platform_contracts.policies import canonical_sha256
from agent_platform_contracts.state_machines import transition, StateTransitionError

class RegistryService:
    """注册中心核心服务"""
    
    def __init__(self, repository: DefinitionRepository):
        self.repo = repository
    
    async def register_definition(self, kind: str, payload: dict) -> dict:
        """注册新的定义对象"""
        # 1. 使用 Pydantic 模型验证
        model_class = RESOURCE_MODELS[kind]
        validated = model_class.model_validate(payload)
        
        # 2. 计算内容摘要
        spec_dict = validated.spec.model_dump()
        content_digest = canonical_sha256(spec_dict)
        
        # 3. 检查是否已存在
        existing = await self.repo.get_by_nkv(
            validated.metadata.namespace,
            validated.metadata.key,
            validated.metadata.version
        )
        if existing:
            raise ValueError("Definition already exists")
        
        # 4. 持久化
        obj_dict = validated.model_dump()
        obj_dict['metadata']['content_digest'] = content_digest
        obj_id = await self.repo.create(obj_dict)
        
        return {"id": obj_id, "content_digest": content_digest}
    
    async def transition_state(self, id: str, target_phase: str, reason: str = None) -> dict:
        """迁移定义对象状态"""
        # 1. 获取当前对象
        obj = await self.repo.get_by_id(id)
        if not obj:
            raise ValueError("Definition not found")
        
        current_phase = obj['status']['phase']
        kind = obj['kind'].lower().replace('-', '_')
        
        # 2. 验证状态迁移
        try:
            result = transition('definition', current_phase, target_phase)
        except StateTransitionError as e:
            raise ValueError(f"Invalid transition: {e}")
        
        # 3. 更新状态
        new_status = {
            'phase': target_phase,
            'observed_revision': obj['revision'],
            'reason': reason
        }
        
        success = await self.repo.update(
            id, 
            obj['revision'],
            {'status': new_status}
        )
        
        if not success:
            raise ValueError("Concurrent modification detected")
        
        return {"id": id, "phase": target_phase}
    
    async def resolve_dependencies(self, id: str, version: str) -> dict:
        """解析依赖关系"""
        # 1. 获取对象
        obj = await self.repo.get_by_nkv(
            namespace='*',  # 从 ID 推断
            key='*',
            version=version
        )
        
        # 2. 递归解析依赖
        dependencies = await self._resolve_recursive(id, version, set())
        
        # 3. 验证所有依赖都是 active 状态
        for dep in dependencies:
            if dep['status']['phase'] != 'active':
                raise ValueError(f"Dependency {dep['id']} is not active")
        
        return {
            "root": obj,
            "dependencies": dependencies,
            "total_count": len(dependencies)
        }
    
    async def _resolve_recursive(self, id: str, version: str, visited: set) -> list:
        """递归解析依赖"""
        key = f"{id}:{version}"
        if key in visited:
            return []
        visited.add(key)
        
        deps = await self.repo.get_dependencies(id, version)
        result = list(deps)
        
        for dep in deps:
            sub_deps = await self._resolve_recursive(
                dep['target_id'], 
                dep['target_version'],
                visited
            )
            result.extend(sub_deps)
        
        return result
```

**学习要点**：
- 服务层协调 Repository 和业务规则
- 状态迁移调用 B0 状态机
- 依赖解析使用递归 + 访问集合防循环

### 任务 4：API 层（第 8-10 天）

#### 2.4.1 FastAPI 路由

```python
# B1-能力注册中心/src/api/routes.py

from fastapi import APIRouter, HTTPException, Query
from pydantic import BaseModel
from typing import Optional, List

router = APIRouter(prefix="/api/v1/registry", tags=["registry"])

class RegisterRequest(BaseModel):
    kind: str
    payload: dict

class TransitionRequest(BaseModel):
    target_phase: str
    reason: Optional[str] = None

@router.post("/definitions")
async def register_definition(request: RegisterRequest):
    """注册新的定义对象"""
    try:
        result = await registry_service.register_definition(
            request.kind, 
            request.payload
        )
        return {"status": "success", "data": result}
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/definitions/{definition_id}")
async def get_definition(definition_id: str):
    """获取定义对象"""
    result = await registry_service.get_definition(definition_id)
    if not result:
        raise HTTPException(status_code=404, detail="Not found")
    return {"status": "success", "data": result}

@router.post("/definitions/{definition_id}/transition")
async def transition_definition(definition_id: str, request: TransitionRequest):
    """迁移定义对象状态"""
    try:
        result = await registry_service.transition_state(
            definition_id,
            request.target_phase,
            request.reason
        )
        return {"status": "success", "data": result}
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/definitions")
async def list_definitions(
    kind: Optional[str] = Query(None),
    namespace: Optional[str] = Query(None),
    phase: Optional[str] = Query(None),
    limit: int = Query(100, ge=1, le=1000)
):
    """查询定义对象列表"""
    filter = DefinitionFilter(
        kind=kind,
        namespace=namespace,
        phase=phase
    )
    results = await registry_service.list_definitions(filter, limit)
    return {"status": "success", "data": results}
```

**学习要点**：
- FastAPI 自动生成 OpenAPI 文档
- Pydantic 模型用于请求验证
- 错误处理返回合适的 HTTP 状态码
