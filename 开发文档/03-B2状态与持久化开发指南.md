## 一、B2 目标

实现运行时对象的持久化管理，包括 Task、AgentRun、Checkpoint、ToolCall、Approval 和 Event。

### 1.1 核心职责

- Task 调度与生命周期管理
- AgentRun 创建、执行与恢复
- Checkpoint 持久化与恢复
- ToolCall 追踪与审批
- Event Store 事件溯源

### 1.2 不包含

- Context 组装（B3）
- Agent 执行逻辑（B4）
- Loop 迭代（B5）

#### 2.2.1 Task 服务

```python
# B2-状态与持久化/src/runtime/task_service.py

from datetime import datetime, timezone
from typing import Optional

class TaskService:
    """Task 调度服务"""
    
    def __init__(self, repository, event_store):
        self.repo = repository
        self.event_store = event_store
    
    async def create_task(self, payload: dict) -> dict:
        """创建新 Task"""
        # 1. 验证幂等键
        existing = await self.repo.find_by_idempotency_key(
            payload['scope']['tenant_id'],
            payload['source']['type'],
            payload['idempotency_key']
        )
        if existing:
            return {"id": existing['id'], "status": "already_exists"}
        
        # 2. 验证 Schema（使用 B0 模型）
        from agent_platform_contracts.models import TaskResource
        validated = TaskResource.model_validate(payload)
        
        # 3. 初始状态：created
        task_dict = validated.model_dump()
        task_dict['status'] = {'phase': 'created'}
        task_dict['created_at'] = datetime.now(timezone.utc).isoformat()
        
        # 4. 持久化
        task_id = await self.repo.create_task(task_dict)
        
        # 5. 发布事件
        await self.event_store.publish(
            event_type="task.created",
            subject_ref={"kind": "Task", "id": task_id},
            data={"task_id": task_id, "title": payload['spec']['title']}
        )
        
        return {"id": task_id, "phase": "created"}
    
    async def validate_task(self, task_id: str) -> dict:
        """验证 Task：Schema、权限、预算"""
        task = await self.repo.get_task(task_id)
        
        # 1. Schema 验证
        # 2. 权限验证
        # 3. 预算验证
        
        # 4. 状态迁移：created -> validated
        await self.repo.update_task_status(task_id, 'validated')
        
        await self.event_store.publish(
            event_type="task.validated",
            subject_ref={"kind": "Task", "id": task_id},
            data={}
        )
        
        return {"id": task_id, "phase": "validated"}
    
    async def queue_task(self, task_id: str) -> dict:
        """将 Task 加入队列"""
        task = await self.repo.get_task(task_id)
        
        # 验证当前状态
        if task['status']['phase'] != 'validated':
            raise ValueError("Task must be validated before queuing")
        
        # 状态迁移：validated -> queued
        await self.repo.update_task_status(task_id, 'queued')
        
        # 通知调度器
        await self._notify_scheduler(task_id)
        
        return {"id": task_id, "phase": "queued"}
    
    async def start_task(self, task_id: str) -> dict:
        """开始执行 Task"""
        task = await self.repo.get_task(task_id)
        
        # 状态迁移：queued -> running
        await self.repo.update_task_status(task_id, 'running')
        
        # 创建 AgentRun
        run_id = await self._create_agent_run(task)
        
        return {"id": task_id, "phase": "running", "run_id": run_id}
    
    async def complete_task(self, task_id: str, artifact_id: str) -> dict:
        """完成 Task"""
        task = await self.repo.get_task(task_id)
        
        # 状态迁移：running -> succeeded
        await self.repo.update_task_status(
            task_id, 
            'succeeded',
            final_artifact_ref={"kind": "Artifact", "id": artifact_id}
        )
        
        await self.event_store.publish(
            event_type="task.succeeded",
            subject_ref={"kind": "Task", "id": task_id},
            data={"artifact_id": artifact_id}
        )
        
        return {"id": task_id, "phase": "succeeded"}
```

#### 2.3.1 AgentRun 服务

```python
# B2-状态与持久化/src/runtime/agent_run_service.py

class AgentRunService:
    """AgentRun 生命周期管理"""
    
    async def create_run(self, task_id: str, agent_spec_id: str) -> dict:
        """创建新的 AgentRun"""
        # 1. 获取 Task
        task = await self.task_repo.get_task(task_id)
        
        # 2. 解析 AgentSpec（从 B1 Registry）
        agent_spec = await self.registry.resolve_definition(agent_spec_id)
        
        # 3. 解析所有依赖
        dependencies = await self._resolve_all_dependencies(agent_spec)
        
        # 4. 创建 Run
        run = {
            "task_id": task_id,
            "attempt": await self._get_next_attempt(task_id),
            "agent_snapshot": {
                "kind": "AgentSpec",
                "id": agent_spec_id,
                "version": agent_spec['metadata']['version'],
                "digest": agent_spec['metadata']['content_digest']
            },
            "resolved_dependencies": dependencies,
            "current_model_selection": dependencies['model_selection'],
            "model_selection_history": [dependencies['model_selection']],
            "budget": task['spec']['budget'],
            "usage": self._init_usage(),
            "status": {"phase": "created"}
        }
        
        run_id = await self.run_repo.create_run(run)
        
        return {"id": run_id, "phase": "created"}
    
    async def resolve_run(self, run_id: str) -> dict:
        """解析 Run 依赖"""
        run = await self.run_repo.get_run(run_id)
        
        # 状态迁移：created -> resolving
        await self.run_repo.update_status(run_id, 'resolving')
        
        # 验证所有依赖都是 active
        deps = run['resolved_dependencies']
        for dep_type, dep_ref in deps.items():
            if isinstance(dep_ref, dict) and 'id' in dep_ref:
                definition = await self.registry.get_definition(dep_ref['id'])
                if definition['status']['phase'] not in ['active', 'deprecated']:
                    raise ValueError(f"Dependency {dep_ref['id']} is not resolvable")
        
        # 状态迁移：resolving -> ready
        await self.run_repo.update_status(run_id, 'ready')
        
        return {"id": run_id, "phase": "ready"}
    
    async def start_run(self, run_id: str, worker_ref: dict) -> dict:
        """开始执行 Run"""
        run = await self.run_repo.get_run(run_id)
        
        # 获取租约
        lease = await self._acquire_lease(run_id, worker_ref)
        
        # 状态迁移：ready -> running
        await self.run_repo.update_status(run_id, 'running', lease=lease)
        
        return {"id": run_id, "phase": "running", "lease": lease}
    
    async def checkpoint_run(self, run_id: str, checkpoint_data: dict) -> dict:
        """创建 Checkpoint"""
        run = await self.run_repo.get_run(run_id)
        
        # 创建 Checkpoint
        checkpoint = {
            "run_id": run_id,
            "sequence": await self._get_next_sequence(run_id),
            "graph_state": checkpoint_data['graph_state'],
            "node_state": checkpoint_data['node_state'],
            "context_digest": checkpoint_data['context_digest'],
            "side_effect_ledger": checkpoint_data.get('side_effect_ledger', []),
            "pending_approvals": checkpoint_data.get('pending_approvals', []),
            "pending_tool_calls": checkpoint_data.get('pending_tool_calls', []),
            "resume_token_ref": checkpoint_data['resume_token_ref'],
            "created_reason": checkpoint_data['reason'],
            "status": {"phase": "committed"}
        }
        
        ckp_id = await self.ckp_repo.create_checkpoint(checkpoint)
        
        # 更新 Run 的 current_checkpoint_ref
        await self.run_repo.update_checkpoint(run_id, ckp_id)
        
        # 状态迁移：running -> checkpointed
        await self.run_repo.update_status(run_id, 'checkpointed')
        
        return {"id": ckp_id, "run_phase": "checkpointed"}
    
    async def resume_run(self, run_id: str, worker_ref: dict) -> dict:
        """从 Checkpoint 恢复 Run"""
        run = await self.run_repo.get_run(run_id)
        
        # 获取最新 Checkpoint
        checkpoint = await self.ckp_repo.get_latest_checkpoint(run_id)
        
        # 验证 Checkpoint 完整性
        await self._validate_checkpoint(checkpoint)
        
        # 获取新租约
        lease = await self._acquire_lease(run_id, worker_ref)
        
        # 状态迁移：checkpointed/suspended -> running
        await self.run_repo.update_status(run_id, 'running', lease=lease)
        
        return {
            "id": run_id, 
            "phase": "running",
            "checkpoint_id": checkpoint['id'],
            "resume_from": checkpoint['node_state']
        }
```

#### 2.4.1 Event Store 实现

```python
# B2-状态与持久化/src/runtime/event_store.py

import uuid
from datetime import datetime, timezone

class EventStore:
    """事件存储"""
    
    def __init__(self, db_connection):
        self.db = db_connection
    
    async def publish(self, event_type: str, subject_ref: dict, data: dict) -> str:
        """发布事件"""
        event_id = f"evt_{uuid.uuid4()}"
        
        # 获取下一个 sequence
        sequence = await self._get_next_sequence(subject_ref['id'])
        
        event = {
            "id": event_id,
            "specversion": "1.0",
            "source": "agent-runtime",
            "type": event_type,
            "time": datetime.now(timezone.utc).isoformat(),
            "datacontenttype": "application/json",
            "subject_ref": subject_ref,
            "sequence": sequence,
            "trace": await self._get_current_trace(),
            "schema_ref": {"id": f"sch_{event_type.replace('.', '_')}", "version": "1.0.0"},
            "data": data,
            "status": {"phase": "recorded"}
        }
        
        # 持久化（只追加）
        await self.db.insert_event(event)
        
        # 发布到消息队列（用于异步消费）
        await self._publish_to_queue(event)
        
        return event_id
    
    async def get_events(self, subject_id: str, after_sequence: int = 0) -> list:
        """获取事件流"""
        return await self.db.query_events(
            subject_id=subject_id,
            after_sequence=after_sequence
        )
    
    async def rebuild_state(self, subject_id: str, initial_state: dict) -> dict:
        """从事件重建状态"""
        events = await self.get_events(subject_id)
        
        state = initial_state.copy()
        for event in events:
            state = self._apply_event(state, event)
        
        return state
    
    def _apply_event(self, state: dict, event: dict) -> dict:
        """应用事件到状态"""
        event_type = event['type']
        data = event['data']
        
        # 根据事件类型更新状态
        if event_type == "task.created":
            state['phase'] = 'created'
        elif event_type == "task.validated":
            state['phase'] = 'validated'
        elif event_type == "task.queued":
            state['phase'] = 'queued'
        # ... 其他事件类型
        
        return state
```

### 任务 5：Worker 租约管理（第 14-16 天）

#### 2.5.1 租约服务

```python
# B2-状态与持久化/src/runtime/lease_service.py

from datetime import datetime, timedelta, timezone

class LeaseService:
    """Worker 租约管理"""
    
    LEASE_DURATION = timedelta(minutes=5)
    HEARTBEAT_INTERVAL = timedelta(minutes=1)
    
    async def acquire_lease(self, run_id: str, worker_ref: dict) -> dict:
        """获取执行租约"""
        # 检查是否有现有租约
        existing = await self.repo.get_active_lease(run_id)
        
        if existing:
            # 检查是否过期
            if datetime.now(timezone.utc) < existing['expires_at']:
                raise ValueError("Run already has active lease")
        
        # 创建新租约
        lease = {
            "lease_id": f"lease_{uuid.uuid4()}",
            "worker_ref": worker_ref,
            "acquired_at": datetime.now(timezone.utc),
            "heartbeat_at": datetime.now(timezone.utc),
            "expires_at": datetime.now(timezone.utc) + self.LEASE_DURATION,
            "fencing_token": await self._get_next_fencing_token(run_id)
        }
        
        await self.repo.save_lease(run_id, lease)
        
        return lease
    
    async def heartbeat(self, run_id: str, lease_id: str) -> dict:
        """续租心跳"""
        lease = await self.repo.get_lease(run_id, lease_id)
        
        if not lease:
            raise ValueError("Lease not found")
        
        # 验证 fencing token
        current_token = await self._get_current_fencing_token(run_id)
        if lease['fencing_token'] != current_token:
            raise ValueError("Stale fencing token")
        
        # 更新心跳
        lease['heartbeat_at'] = datetime.now(timezone.utc)
        lease['expires_at'] = datetime.now(timezone.utc) + self.LEASE_DURATION
        
        await self.repo.save_lease(run_id, lease)
        
        return lease
    
    async def release_lease(self, run_id: str, lease_id: str) -> None:
        """释放租约"""
        await self.repo.delete_lease(run_id, lease_id)
    
    async def check_lease_valid(self, run_id: str, fencing_token: int) -> bool:
        """检查租约是否有效"""
        lease = await self.repo.get_active_lease(run_id)
        
        if not lease:
            return False
        
        from agent_platform_contracts.policies import lease_allows_commit
        return lease_allows_commit(
            fencing_token=fencing_token,
            current_fencing_token=lease['fencing_token'],
            expires_at=lease['expires_at']
        )
```

**学习要点**：
- 租约防止多个 Worker 同时执行同一 Run
- Fencing Token 防止过期 Worker 提交结果
- 心跳机制检测 Worker 存活
